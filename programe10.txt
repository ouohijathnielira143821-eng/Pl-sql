DECLARE
    v_eid EMP.EID%TYPE;
BEGIN
    v_eid := &EID;

    DELETE FROM EMP
    WHERE EID = v_eid;

    IF SQL%ROWCOUNT = 0 THEN
        DBMS_OUTPUT.PUT_LINE('Employee not found.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Employee record deleted successfully.');
        COMMIT;
    END IF;
END;
/

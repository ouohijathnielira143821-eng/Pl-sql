DECLARE
    v_count NUMBER;
BEGIN
    UPDATE EMP
    SET SAL = SAL * 1.10
    WHERE DEPTNO = 10;

    v_count := SQL%ROWCOUNT;

    IF v_count > 0 THEN
        DBMS_OUTPUT.PUT_LINE(
            v_count || ' employee(s) received a 10% salary increase.'
        );
    ELSE
        DBMS_OUTPUT.PUT_LINE(
            'No employees found in department 10.'
        );
    END IF;
END;
/

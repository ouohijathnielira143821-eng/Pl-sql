DECLARE
    CURSOR c_emp IS
        SELECT EMPNO, ENAME, SAL
        FROM EMP
        WHERE DEPTNO = 20;

    v_empno EMP.EMPNO%TYPE;
    v_ename EMP.ENAME%TYPE;
    v_sal   EMP.SAL%TYPE;

    v_count NUMBER := 0;
BEGIN
    OPEN c_emp;

    IF c_emp%ISOPEN THEN
        DBMS_OUTPUT.PUT_LINE('Explicit cursor is open.');
    END IF;

    LOOP
        FETCH c_emp INTO v_empno, v_ename, v_sal;

        EXIT WHEN c_emp%NOTFOUND;

        UPDATE EMP
        SET SAL = SAL * 1.05
        WHERE EMPNO = v_empno;

        INSERT INTO EMP_UPDATE
            (EMPNO, ENAME, OLD_SAL, NEW_SAL, UPDATE_DATE)
        VALUES
            (v_empno, v_ename, v_sal, v_sal * 1.05, SYSDATE);

        v_count := v_count + 1;
    END LOOP;

    CLOSE c_emp;

    IF v_count > 0 THEN
        DBMS_OUTPUT.PUT_LINE(
            v_count || ' employee(s) received a 5% salary increase.'
        );
    ELSE
        DBMS_OUTPUT.PUT_LINE(
            'No employees found in department 20.'
        );
    END IF;

    COMMIT;
END;
/

DECLARE
    v_count NUMBER;
BEGIN
    UPDATE EMP
    SET SAL = SAL * 1.05
    WHERE DEPTNO = 20;

    v_count := SQL%ROWCOUNT;

    IF SQL%ISOPEN THEN
        DBMS_OUTPUT.PUT_LINE('Implicit cursor is open.');
    ELSE
        DBMS_OUTPUT.PUT_LINE('Implicit cursor is closed.');
    END IF;

    IF SQL%NOTFOUND THEN
        DBMS_OUTPUT.PUT_LINE(
            'No employees found in department 20.'
        );
    ELSE
        DBMS_OUTPUT.PUT_LINE(
            v_count || ' employee(s) received a 5% salary increase.'
        );

        INSERT INTO EMP_UPDATE
        SELECT EMPNO, ENAME, SAL / 1.05, SAL, SYSDATE
        FROM EMP
        WHERE DEPTNO = 20;
    END IF;

    COMMIT;
END;

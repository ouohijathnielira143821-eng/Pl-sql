BEGIN
    FOR emp_rec IN (
        SELECT *
        FROM EMP
        WHERE Gender = 'M'
          AND Deptname = 'HR'
    )
    LOOP
        DBMS_OUTPUT.PUT_LINE(
            'EID: ' || emp_rec.EID ||
            ', Name: ' || emp_rec.EName ||
            ', Dept No: ' || emp_rec.Deptno ||
            ', Dept: ' || emp_rec.Deptname ||
            ', Gender: ' || emp_rec.Gender ||
            ', Age: ' || emp_rec.Age ||
            ', Basic Salary: ' || emp_rec.BasicSal
        );
    END LOOP;
END;
/

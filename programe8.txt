DECLARE
    v_eid       EMP.EID%TYPE;
    v_basic     EMP.BasicSal%TYPE;
    v_gross     NUMBER(10,2);
    v_da        NUMBER(10,2);
    v_hra       NUMBER(10,2);
    v_medical   NUMBER(10,2);
    v_pf        NUMBER(10,2);
BEGIN
    v_eid := &EID;

    SELECT BasicSal
    INTO v_basic
    FROM EMP
    WHERE EID = v_eid;

    v_da      := v_basic * 0.50;
    v_hra     := v_basic * 0.15;
    v_medical := 500;
    v_pf      := v_basic * 0.10;

    v_gross := v_basic + (v_da + v_hra + v_medical) - v_pf;

    DBMS_OUTPUT.PUT_LINE('Gross Salary = Rs. ' || v_gross);

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('Employee not found.');
END;
/

SET SERVEROUTPUT ON;

DECLARE
    V_A NUMBER := 100;
    V_B NUMBER := 0;
    V_RESULT NUMBER;

BEGIN

    V_RESULT := V_A / V_B;

EXCEPTION

    WHEN OTHERS THEN
        DBMS_OUTPUT.PUT_LINE(
            'Error Code    : ' || SQLCODE
        );

        DBMS_OUTPUT.PUT_LINE(
            'Error Message : ' || SQLERRM
        );

END;
/

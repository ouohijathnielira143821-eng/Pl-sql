SET SERVEROUTPUT ON;

DECLARE
    V_A NUMBER := &Enter_First_Number;
    V_B NUMBER := &Enter_Second_Number;
    V_RESULT NUMBER;

BEGIN

    V_RESULT := V_A / V_B;

    DBMS_OUTPUT.PUT_LINE(
        'Result = ' || V_RESULT
    );

EXCEPTION

    WHEN ZERO_DIVIDE THEN
        DBMS_OUTPUT.PUT_LINE(
            'Error: Second number cannot be zero.'
        );

END;
/
